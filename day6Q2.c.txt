#include <stdio.h>

int main()
{
    int n;

    printf("Enter the value of n: ");
    scanf("%d", &n);

    if (n > 0)
    {
        printf("Positive integer");
    }
    else
    {
        if (n < 0)
        {
            printf("Negative integer");
        }
        else
        {
            printf("Zero");
        }
    }

    return 0;
}
